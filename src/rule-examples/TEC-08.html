TypeScript, CodeMirror 6
import { HighlightStyle } from "@codemirror/language";
import { tags as t } from "@lezer/highlight";

/** sebaro code looks (spec/01 §1.8): three tones, comments grey, the rest ink; no bold or italic in code. */
export const sebaroHighlight = HighlightStyle.define([
  { tag: [t.keyword, t.controlKeyword, t.definitionKeyword, t.moduleKeyword, t.operatorKeyword, t.modifier, t.self,
          t.annotation, t.meta, t.macroName, t.tagName], color: "var(--sb-syntax-keyword)" },
  { tag: [t.string, t.special(t.string), t.regexp, t.character, t.attributeValue, t.docString],
    color: "var(--sb-syntax-string)" },
  { tag: [t.number, t.integer, t.float, t.bool, t.null, t.atom, t.unit, t.color, t.escape, t.constant(t.variableName)],
    color: "var(--sb-syntax-constant)" },
  { tag: [t.comment, t.lineComment, t.blockComment, t.docComment], color: "var(--sb-color-text-muted)" },
  { tag: t.invalid, textDecoration: "underline wavy var(--sb-color-danger) 2px" },
]);
